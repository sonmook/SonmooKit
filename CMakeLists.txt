FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
function(add_effect target code kind category)
    if(kind LESS_EQUAL 3 OR kind EQUAL 6 OR kind EQUAL 7 OR kind EQUAL 8)
        set(effect_options VERSION 0.12.1)
    elseif(ARGV4)
        set(effect_options VERSION 0.13.2 PRODUCT_NAME "Dial ${ARGV4}" PLUGIN_NAME "Dial ${ARGV4}")
    elseif(kind EQUAL 9)
        set(effect_options NEEDS_MIDI_INPUT TRUE PRODUCT_NAME "SonmooTune" VERSION 0.12.0)
    elseif(kind EQUAL 10)
        set(effect_options VERSION 0.10.1)
    elseif(kind EQUAL 11)
        set(effect_options VERSION 0.11.0)
    endif()
    juce_add_plugin(${target}
        COMPANY_NAME "SonmooKit"
        BUNDLE_ID "org.audiokit.${target}"
        PLUGIN_MANUFACTURER_CODE Akfx
        PLUGIN_CODE ${code}
        ${effect_options}
        FORMATS VST3
        VST3_CATEGORIES Fx ${category}
        COPY_PLUGIN_AFTER_BUILD FALSE)
    if(kind GREATER_EQUAL 12)
        target_sources(${target} PRIVATE Macro.cpp)
    elseif(kind GREATER_EQUAL 5 AND kind LESS_EQUAL 7)
        target_sources(${target} PRIVATE Advanced.cpp)
        target_link_libraries(${target} PRIVATE juce::juce_dsp)
    elseif(kind EQUAL 9)
        target_sources(${target} PRIVATE PitchPilot.cpp)
    else()
        target_sources(${target} PRIVATE Plugin.cpp)
    endif()
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE EFFECT_KIND=${kind}
        JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(${target} PRIVATE juce::juce_audio_processors)
endfunction()
